### Step 4 — Auto-absent background job in `server/index.ts`

Add a new function `runAbsentMarkingJob()` and schedule it with `setInterval` every 60 seconds (same pattern as the existing factory/overtime notification crons).

The function should run once per day at **23:55 Cairo time** using the existing `lastRun` map and `shouldRunAt()` helper.

Logic inside `runAbsentMarkingJob()`:

```ts
import { factoryRotationSchedule, overtimeRotationSchedule } from '../shared/schema'
import { isNull, eq, and, lt } from 'drizzle-orm'

async function runAbsentMarkingJob() {
  try {
    const now = new Date()
    const yesterday = new Date(now)
    yesterday.setDate(yesterday.getDate() - 1)
    const yesterdayStr = yesterday.toLocaleDateString('en-CA', { timeZone: 'Africa/Cairo' })

    // Mark factory absences
    const factoryAbsent = await db
      .select({ id: factoryRotationSchedule.id, user_id: factoryRotationSchedule.user_id, scheduled_date: factoryRotationSchedule.scheduled_date })
      .from(factoryRotationSchedule)
      .where(and(
        eq(factoryRotationSchedule.scheduled_date, yesterdayStr),
        isNull(factoryRotationSchedule.attended_at),
        eq(factoryRotationSchedule.is_absent, false),
      ))

    for (const row of factoryAbsent) {
      await db.update(factoryRotationSchedule)
        .set({ is_absent: true })
        .where(eq(factoryRotationSchedule.id, row.id))

      await db.insert(notifications).values({
        user_id: row.user_id,
        message: `❌ تم تسجيل غيابك في المصنع بتاريخ ${yesterdayStr}`,
      })
      broadcast(row.user_id, 'notification', { message: `❌ تم تسجيل غيابك في المصنع بتاريخ ${yesterdayStr}` })
    }

    // Mark overtime absences — same logic
    const overtimeAbsent = await db
      .select({ id: overtimeRotationSchedule.id, user_id: overtimeRotationSchedule.user_id, scheduled_date: overtimeRotationSchedule.scheduled_date })
      .from(overtimeRotationSchedule)
      .where(and(
        eq(overtimeRotationSchedule.scheduled_date, yesterdayStr),
        isNull(overtimeRotationSchedule.attended_at),
        eq(overtimeRotationSchedule.is_absent, false),
      ))

    for (const row of overtimeAbsent) {
      await db.update(overtimeRotationSchedule)
        .set({ is_absent: true })
        .where(eq(overtimeRotationSchedule.id, row.id))

      await db.insert(notifications).values({
        user_id: row.user_id,
        message: `❌ تم تسجيل غيابك في الأوفرتايم بتاريخ ${yesterdayStr}`,
      })
      broadcast(row.user_id, 'notification', { message: `❌ تم تسجيل غيابك في الأوفرتايم بتاريخ ${yesterdayStr}` })
    }

    // Notify admins summary
    const absentNames: string[] = []
    if (factoryAbsent.length || overtimeAbsent.length) {
      const allAbsentUserIds = [...new Set([...factoryAbsent, ...overtimeAbsent].map(r => r.user_id))]
      const empRows = await db.select({ id: profiles.id, full_name: profiles.full_name, email: profiles.email })
        .from(profiles).where(inArray(profiles.id, allAbsentUserIds))
      const nameMap = Object.fromEntries(empRows.map(e => [e.id, e.full_name || e.email]))

      const factoryNames = factoryAbsent.map(r => nameMap[r.user_id]).filter(Boolean)
      const overtimeNames = overtimeAbsent.map(r => nameMap[r.user_id]).filter(Boolean)

      const adminRows = await db.select({ id: profiles.id }).from(profiles).where(inArray(profiles.role, ['admin', 'super_admin']))

      let adminMsg = `📋 تقرير الغياب ليوم ${yesterdayStr}\n`
      if (factoryNames.length) adminMsg += `🏭 مصنع: ${factoryNames.join('، ')}\n`
      if (overtimeNames.length) adminMsg += `⏱️ أوفرتايم: ${overtimeNames.join('، ')}`

      for (const admin of adminRows) {
        await db.insert(notifications).values({ user_id: admin.id, message: adminMsg })
        broadcast(admin.id, 'notification', { message: adminMsg })
      }
    }

    console.log(`[AbsentJob] Factory: ${factoryAbsent.length}, Overtime: ${overtimeAbsent.length} marked absent for ${yesterdayStr}`)
  } catch (err) {
    console.error('[AbsentJob error]', err)
  }
}
```

```ts
router.post('/', requireAuth, async (req, res) => {
  const { module, schedule_id, target_user_id, target_schedule_id, requester_note } = req.body
  if (!module || !schedule_id || !target_user_id)
    return res.status(400).json({ error: 'module, schedule_id, target_user_id are required' })

  // Verify schedule_id belongs to requester and is in the future
  const table = module === 'factory' ? factoryRotationSchedule : overtimeRotationSchedule
  const [entry] = await db.select().from(table).where(eq(table.id, schedule_id)).limit(1)
  if (!entry) return res.status(404).json({ error: 'Schedule entry not found' })
  if (entry.user_id !== req.user.id) return res.status(403).json({ error: 'هذا اليوم ليس يومك' })

  const today = new Date().toLocaleDateString('en-CA', { timeZone: 'Africa/Cairo' })
  const entryDate = String(entry.scheduled_date).slice(0, 10)
  if (entryDate <= today) return res.status(400).json({ error: 'لا يمكن طلب تحويل يوم مضى' })

  // Check no existing pending swap for this schedule_id
  const existing = await db.select().from(rotationSwapRequests)
    .where(and(eq(rotationSwapRequests.schedule_id, schedule_id), eq(rotationSwapRequests.status, 'pending')))
    .limit(1)
  if (existing.length) return res.status(400).json({ error: 'يوجد طلب تحويل معلق لهذا اليوم بالفعل' })

  const [swap] = await db.insert(rotationSwapRequests).values({
    module, schedule_id, requester_id: req.user.id,
    target_user_id, target_schedule_id: target_schedule_id || null,
    requester_note: requester_note || null,
  }).returning()

  // Notify target
  const [requester] = await db.select({ full_name: profiles.full_name }).from(profiles).where(eq(profiles.id, req.user.id)).limit(1)
  const requesterName = requester?.full_name || 'موظف'
  const moduleName = module === 'factory' ? 'المصنع' : 'الأوفرتايم'
  const msg = `🔄 ${requesterName} يطلب تحويل يوم دورته في ${moduleName} بتاريخ ${entryDate} إليك — يرجى الرد من صفحة التناوب`

  await db.insert(notifications).values({ user_id: target_user_id, message: msg })
  broadcast(target_user_id, 'notification', { message: msg })

  res.json(swap)
})
```

```ts
router.get('/my', requireAuth, async (req, res) => {
  const sent = await db.select().from(rotationSwapRequests)
    .where(eq(rotationSwapRequests.requester_id, req.user.id))
    .orderBy(desc(rotationSwapRequests.created_at))

  const received = await db.select().from(rotationSwapRequests)
    .where(eq(rotationSwapRequests.target_user_id, req.user.id))
    .orderBy(desc(rotationSwapRequests.created_at))

  // Enrich with user names
  const allUserIds = [...new Set([
    ...sent.map(s => s.target_user_id),
    ...received.map(r => r.requester_id),
  ])]
  const users = allUserIds.length
    ? await db.select({ id: profiles.id, full_name: profiles.full_name }).from(profiles).where(inArray(profiles.id, allUserIds))
    : []
  const nameMap = Object.fromEntries(users.map(u => [u.id, u.full_name]))

  // Enrich with scheduled dates
  const allScheduleIds = [...new Set([
    ...sent.map(s => s.schedule_id),
    ...received.map(r => r.schedule_id),
    ...sent.filter(s => s.target_schedule_id).map(s => s.target_schedule_id!),
    ...received.filter(r => r.target_schedule_id).map(r => r.target_schedule_id!),
  ])]

  // Get dates from both tables
  const factoryDates = await db.select({ id: factoryRotationSchedule.id, scheduled_date: factoryRotationSchedule.scheduled_date })
    .from(factoryRotationSchedule).where(inArray(factoryRotationSchedule.id, allScheduleIds))
  const overtimeDates = await db.select({ id: overtimeRotationSchedule.id, scheduled_date: overtimeRotationSchedule.scheduled_date })
    .from(overtimeRotationSchedule).where(inArray(overtimeRotationSchedule.id, allScheduleIds))
  const dateMap = Object.fromEntries([...factoryDates, ...overtimeDates].map(r => [r.id, r.scheduled_date]))

  const enrich = (list: any[], type: 'sent' | 'received') => list.map(s => ({
    ...s,
    type,
    target_name: nameMap[s.target_user_id],
    requester_name: nameMap[s.requester_id],
    scheduled_date: dateMap[s.schedule_id],
    target_scheduled_date: s.target_schedule_id ? dateMap[s.target_schedule_id] : null,
  }))

  res.json({ sent: enrich(sent, 'sent'), received: enrich(received, 'received') })
})
```

```ts
router.post('/:id/accept', requireAuth, async (req, res) => {
  const [swap] = await db.select().from(rotationSwapRequests).where(eq(rotationSwapRequests.id, req.params.id)).limit(1)
  if (!swap) return res.status(404).json({ error: 'Swap request not found' })

  if (swap.target_schedule_id) {
    await db.update(table).set({ user_id: swap.requester_id }).where(eq(table.id, swap.target_schedule_id))
  }

  await db.update(rotationSwapRequests)
    .set({ status: 'accepted', resolved_at: new Date() })
    .where(eq(rotationSwapRequests.id, swap.id))

  // Notify requester
  const [accepter] = await db.select({ full_name: profiles.full_name }).from(profiles).where(eq(profiles.id, req.user.id)).limit(1)

  // Notify admins
  const admins = await db.select({ id: profiles.id }).from(profiles).where(inArray(profiles.role, ['admin', 'super_admin']))

  const adminMsg = `🔄 تم تحويل يوم التناوب في ${moduleName} بين ${requesterProfile?.full_name || 'موظف'} و ${accepter?.full_name || 'موظف'}`
```
