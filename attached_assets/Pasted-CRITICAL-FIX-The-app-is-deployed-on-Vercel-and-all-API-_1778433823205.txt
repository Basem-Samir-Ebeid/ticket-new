CRITICAL FIX — The app is deployed on Vercel and all API requests go through `api/index.js`, NOT `server/routes/users.ts`. The "No fields to update" error when saving whatsapp_phone comes from `api/index.js` because its `PATCH /api/users/:id` handler does NOT include `whatsapp_phone`.

## Fix: Update `PATCH /api/users/:id` in `api/index.js`

Find this handler in `api/index.js` (around line 672):

```js
app.patch('/api/users/:id', requireAuth, requireAdmin, async (req, res) => {
  try {
    const { full_name, role, can_view_attendance, profile_picture_url, leave_balance } = req.body
    const fields = [], values = []
    let idx = 1
    if (full_name !== undefined) { fields.push(`full_name=$${idx++}`); values.push(full_name) }
    if (role !== undefined) { fields.push(`role=$${idx++}`); values.push(role) }
    if (can_view_attendance !== undefined) { fields.push(`can_view_attendance=$${idx++}`); values.push(can_view_attendance) }
    if (profile_picture_url !== undefined) { fields.push(`profile_picture_url=$${idx++}`); values.push(profile_picture_url) }
    if (leave_balance !== undefined) { fields.push(`leave_balance=$${idx++}`); values.push(leave_balance) }
    if (!fields.length) return res.status(400).json({ error: 'No fields to update' })
    values.push(req.params.id)

      `UPDATE profiles SET ${fields.join(',')} WHERE id=$${idx} RETURNING id, email, full_name, role, can_view_attendance, profile_picture_url, must_change_password, leave_balance, created_at`,

    console.error('[PATCH /users/:id] Error:', err.message)
    res.status(500).json({ error: 'Failed to update user' })
```

Replace it ENTIRELY with this updated version:

```js
app.patch('/api/users/:id', requireAuth, requireAdmin, async (req, res) => {
  try {
    const {
      full_name, role, can_view_attendance, profile_picture_url,
      leave_balance, sick_leave_balance, emergency_leave_balance, work_start_hour,
      department, job_title, phone, national_id, hire_date, birth_date,
      gender, address, employment_type, employee_code, direct_manager, notes,
      whatsapp_phone,
    } = req.body

    const fields = [], values = []
    let idx = 1

    if (full_name !== undefined)           { fields.push(`full_name=$${idx++}`);            values.push(full_name) }
    if (role !== undefined)                { fields.push(`role=$${idx++}`);                 values.push(role) }
    if (can_view_attendance !== undefined)  { fields.push(`can_view_attendance=$${idx++}`);  values.push(can_view_attendance) }
    if (profile_picture_url !== undefined)  { fields.push(`profile_picture_url=$${idx++}`);  values.push(profile_picture_url) }
    if (leave_balance !== undefined)        { fields.push(`leave_balance=$${idx++}`);         values.push(Number(leave_balance)) }
    if (sick_leave_balance !== undefined)   { fields.push(`sick_leave_balance=$${idx++}`);    values.push(Number(sick_leave_balance)) }
    if (emergency_leave_balance !== undefined) { fields.push(`emergency_leave_balance=$${idx++}`); values.push(Number(emergency_leave_balance)) }
    if (work_start_hour !== undefined)      { fields.push(`work_start_hour=$${idx++}`);      values.push(Number(work_start_hour)) }
    if (department !== undefined)           { fields.push(`department=$${idx++}`);           values.push(department || null) }
    if (job_title !== undefined)            { fields.push(`job_title=$${idx++}`);            values.push(job_title || null) }
    if (phone !== undefined)               { fields.push(`phone=$${idx++}`);               values.push(phone || null) }
    if (national_id !== undefined)          { fields.push(`national_id=$${idx++}`);          values.push(national_id || null) }
    if (hire_date !== undefined)            { fields.push(`hire_date=$${idx++}`);            values.push(hire_date || null) }
    if (birth_date !== undefined)           { fields.push(`birth_date=$${idx++}`);           values.push(birth_date || null) }
    if (gender !== undefined)              { fields.push(`gender=$${idx++}`);              values.push(gender || null) }
    if (address !== undefined)             { fields.push(`address=$${idx++}`);             values.push(address || null) }
    if (employment_type !== undefined)      { fields.push(`employment_type=$${idx++}`);      values.push(employment_type || null) }
    if (employee_code !== undefined)        { fields.push(`employee_code=$${idx++}`);        values.push(employee_code || null) }
    if (direct_manager !== undefined)       { fields.push(`direct_manager=$${idx++}`);       values.push(direct_manager || null) }
    if (notes !== undefined)               { fields.push(`notes=$${idx++}`);               values.push(notes || null) }
    if (whatsapp_phone !== undefined)       { fields.push(`whatsapp_phone=$${idx++}`);       values.push(whatsapp_phone?.trim() || null) }

    if (!fields.length) return res.status(400).json({ error: 'No fields to update' })

    values.push(req.params.id)
    const { rows } = await getPool().query(
      `UPDATE profiles SET ${fields.join(', ')} WHERE id=$${idx} RETURNING *`,
      values
    )
    if (!rows[0]) return res.status(404).json({ error: 'User not found' })

    // Strip sensitive fields before returning
    const { password_hash, whatsapp_apikey, plain_password, ...safeUser } = rows[0]
    res.json(safeUser)
  } catch (err) {
    console.error('[PATCH /users/:id] Error:', err.message)
    res.status(500).json({ error: err.message || 'Failed to update user' })
  }
})
```

---

This is the ONLY change needed. The root cause is that `api/index.js` is what Vercel actually runs, and its `PATCH /api/users/:id` never handled `whatsapp_phone` — so sending `{ whatsapp_phone: '01023588751' }` always triggered "No fields to update".
