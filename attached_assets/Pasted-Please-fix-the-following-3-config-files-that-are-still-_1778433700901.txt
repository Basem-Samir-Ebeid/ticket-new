## FIX 2 — `server/smtpConfig.ts` — Saves SMTP settings to `smtp-config.json` only

SMTP config (host, port, user, password, from) is saved to `server/smtp-config.json` via `fs.writeFileSync`. This file is lost on Replit restart, causing email notifications to stop working.

**Fix:** Migrate SMTP config to the database using the existing `systemSettings` table (same table used by `whatsappConfig.ts`).

Replace the file-based `getSmtpConfig()` and `saveSmtpConfig()` with DB-backed versions:

```ts
import { db } from './db'
import { systemSettings } from '../shared/schema'
import { eq } from 'drizzle-orm'

const SMTP_KEY = 'smtp_config'

export interface SmtpConfig {
  host: string
  port: number
  user: string
  password: string
  from: string
  enabled: boolean
}

const DEFAULT_SMTP: SmtpConfig = {
  host: '',
  port: 587,
  user: '',
  password: '',
  from: '',
  enabled: false,
}

export async function getSmtpConfig(): Promise<SmtpConfig> {
  try {
    const [row] = await db.select({ value: systemSettings.value })
      .from(systemSettings)
      .where(eq(systemSettings.key, SMTP_KEY))
      .limit(1)
    if (row?.value) return { ...DEFAULT_SMTP, ...JSON.parse(row.value) }
  } catch {}
  return { ...DEFAULT_SMTP }
}

export async function saveSmtpConfig(config: Partial<SmtpConfig>): Promise<SmtpConfig> {
  const existing = await getSmtpConfig()
  const merged: SmtpConfig = { ...existing, ...config }
  await db.insert(systemSettings)
    .values({ key: SMTP_KEY, value: JSON.stringify(merged) })
    .onConflictDoUpdate({ target: systemSettings.key, set: { value: JSON.stringify(merged), updated_at: new Date() } })
  return merged
}
```

After migrating, delete the file `server/smtp-config.json` from the project.

**Important:** All callers of `getSmtpConfig()` and `saveSmtpConfig()` must be updated to `await` them since they are now async. Check `server/mailer.ts` and any settings routes that call these functions and add `await` where missing.
